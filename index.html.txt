  <script>
    (function() {
      // ========== 配置：替换成你自己的JSONBin信息 ==========
      const JSONBIN_BIN_ID = '你的BIN_ID';          // 例如 '64abc123...'
      const JSONBIN_API_KEY = '你的ACCESS_KEY';     // 在 jsonbin.io 的 API Keys 页面获取
      const JSONBIN_URL = `https://api.jsonbin.io/v3/b/${JSONBIN_BIN_ID}`;
      const ADMIN_PASSWORD = 'admin888';           // 管理员密码，可自行修改

      // ========== 其他常量 ==========
      const TIME_SLOTS = [
        { id: 'morning', label: '上午', timeRange: '9:00-12:00', startHour: 9, icon: '☀️' },
        { id: 'afternoon', label: '下午', timeRange: '15:00-18:00', startHour: 15, icon: '🌤️' },
        { id: 'evening', label: '晚上', timeRange: '19:00-22:00', startHour: 19, icon: '🌙' },
      ];
      const WEEKDAY_NAMES = ['周日','周一','周二','周三','周四','周五','周六'];
      let currentWeekMonday = getMondayOfWeek(new Date());
      let selectedSlotKey = null;
      let isAdminLoggedIn = false;
      let localCache = {}; // 缓存从JSONBin拉取的最新数据

      function getMondayOfWeek(d){const r=new Date(d);r.setHours(0,0,0,0);const day=r.getDay();const diff=day===0?-6:1-day;r.setDate(r.getDate()+diff);return r;}
      function formatDateKey(d){return `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}-${String(d.getDate()).padStart(2,'0')}`}
      function formatDateDisplay(d){return `${d.getMonth()+1}月${d.getDate()}日`}
      function isSlotExpired(date,startHour){return new Date() >= new Date(date.getFullYear(),date.getMonth(),date.getDate(),startHour,0,0);}

      // ========== 与JSONBin交互 ==========
      async function fetchRemoteData(){
        const resp = await fetch(JSONBIN_URL, { headers: { 'X-Master-Key': JSONBIN_API_KEY } });
        if(!resp.ok) throw new Error('数据读取失败');
        const json = await resp.json();
        localCache = json.record;
        return localCache;
      }

      async function updateRemoteData(newData){
        const resp = await fetch(JSONBIN_URL, {
          method: 'PUT',
          headers: { 'Content-Type': 'application/json', 'X-Master-Key': JSONBIN_API_KEY },
          body: JSON.stringify(newData)
        });
        if(!resp.ok) throw new Error('数据更新失败');
        localCache = newData;
      }

      function getSlotData(dateKey, slotId){ return localCache[dateKey]?.[slotId] || null; }
      async function bookSlot(dateKey, slotId, bookingInfo){
        // 再次拉取最新数据，避免覆盖他人写入
        await fetchRemoteData();
        if(getSlotData(dateKey, slotId)) throw new Error('时段已被占用');
        if(!localCache[dateKey]) localCache[dateKey] = {};
        localCache[dateKey][slotId] = bookingInfo;
        await updateRemoteData(localCache);
      }
      async function deleteSlot(dateKey, slotId){
        await fetchRemoteData();
        if(localCache[dateKey] && localCache[dateKey][slotId]){
          delete localCache[dateKey][slotId];
          if(Object.keys(localCache[dateKey]).length===0) delete localCache[dateKey];
          await updateRemoteData(localCache);
        }
      }

      // ========== 渲染 ==========
      function renderScheduleTable(){
        const weekDates = getWeekDates(currentWeekMonday);
        const startDisp = formatDateDisplay(weekDates[0]), endDisp = formatDateDisplay(weekDates[6]);
        document.getElementById('weekLabel').textContent = `${weekDates[0].getFullYear()}年 · ${startDisp} — ${endDisp}`;
        let headerHTML = '<th>时段</th>';
        weekDates.forEach(d=>{headerHTML+=`<th>${d.getDate()}<br><small>${d.getMonth()+1}月 ${WEEKDAY_NAMES[d.getDay()]}</small></th>`;});
        document.getElementById('tableHeaderRow').innerHTML = headerHTML;
        let bodyHTML='';
        TIME_SLOTS.forEach(slot=>{
          bodyHTML+='<tr><td class="time-label">'+slot.icon+' '+slot.label+'<br><small>'+slot.timeRange+'</small></td>';
          weekDates.forEach(d=>{
            const dk=formatDateKey(d);
            const data=getSlotData(dk,slot.id);
            const expired=isSlotExpired(d,slot.startHour);
            let cls='',icon='',text='';
            if(expired){cls='expired';icon='⏰';text='已过期';}
            else if(data){cls='booked';icon='🔒';text='已预约';}
            else{cls='free';icon='✨';text='点击预约';}
            bodyHTML+=`<td class="slot-cell ${cls}" ${cls==='free'?`data-date-key="${dk}" data-slot-id="${slot.id}"`:''}>${icon} ${text}</td>`;
          });
          bodyHTML+='</tr>';
        });
        document.querySelector('#scheduleTable tbody').innerHTML = bodyHTML;
        document.querySelectorAll('.slot-cell.free').forEach(cell=>cell.addEventListener('click',()=>openBookingModal(cell.dataset.dateKey, cell.dataset.slotId)));
      }

      function getWeekDates(monday){const arr=[];for(let i=0;i<7;i++){const d=new Date(monday);d.setDate(d.getDate()+i);arr.push(d);}return arr;}

      function openBookingModal(dk,sid){
        if(getSlotData(dk,sid)){showToast('时段已被预约','error');renderScheduleTable();return;}
        const [y,m,d]=dk.split('-');const date=new Date(y,m-1,d);
        const slot=TIME_SLOTS.find(s=>s.id===sid);
        if(slot&&isSlotExpired(date,slot.startHour)){showToast('已过期','error');return;}
        selectedSlotKey=dk+'|'+sid;
        document.getElementById('selectedSlotInfo').textContent=`📅 ${formatDateDisplay(date)} ${WEEKDAY_NAMES[date.getDay()]} · ${slot.icon} ${slot.label} ${slot.timeRange}`;
        document.getElementById('modalOverlay').style.display='flex';
      }

      document.getElementById('btnCancelBooking').addEventListener('click',()=>{document.getElementById('modalOverlay').style.display='none';selectedSlotKey=null;});
      document.getElementById('bookingForm').addEventListener('submit', async function(e){
        e.preventDefault();
        if(!selectedSlotKey) return;
        const [dk,sid]=selectedSlotKey.split('|');
        const wechat=document.getElementById('wechat').value.trim();
        const birthDate=document.getElementById('birthDate').value;
        const birthTime=document.getElementById('birthTime').value;
        const birthPlace=document.getElementById('birthPlace').value.trim();
        const online=document.querySelector('input[name="online"]:checked')?.value;
        if(!wechat||!birthDate||!birthTime||!birthPlace||!online){showToast('请填写完整信息','error');return;}
        try{
          await bookSlot(dk,sid,{wechat,birthDate,birthTime,birthPlace,online,timestamp:new Date().toISOString()});
          document.getElementById('modalOverlay').style.display='none';
          selectedSlotKey=null;
          renderScheduleTable();
          showToast('预约成功！','success');
        }catch(err){showToast(err.message,'error');}
      });

      // 管理面板
      document.getElementById('btnAdminTrigger').addEventListener('click',()=>{document.getElementById('adminPanel').style.display='block';});
      document.getElementById('btnAdminClose').addEventListener('click',()=>{document.getElementById('adminPanel').style.display='none';});
      document.getElementById('btnAdminLogin').addEventListener('click',()=>{
        if(document.getElementById('adminPasswordInput').value===ADMIN_PASSWORD){
          isAdminLoggedIn=true;document.getElementById('adminContent').style.display='block';
          renderAdminTable();
        }else{showToast('密码错误','error');}
      });
      function renderAdminTable(){
        const rows=[];
        getWeekDates(currentWeekMonday).forEach(d=>{
          const dk=formatDateKey(d);
          TIME_SLOTS.forEach(slot=>{
            const data=getSlotData(dk,slot.id);
            if(data) rows.push({date:formatDateDisplay(d),week:WEEKDAY_NAMES[d.getDay()],slot:slot.label+' '+slot.timeRange,...data,dk,slotId:slot.id});
          });
        });
        const tbody=document.getElementById('adminTableBody');
        tbody.innerHTML=rows.length?rows.map(r=>`<tr><td>${r.date} ${r.week}</td><td>${r.slot}</td><td>${r.wechat}</td><td>${r.birthDate} ${r.birthTime}</td><td>${r.birthPlace}</td><td>${r.online}</td><td><button class="btn-cancel" data-dk="${r.dk}" data-sid="${r.slotId}">取消</button></td></tr>`).join(''):'<tr><td colspan="7">暂无记录</td></tr>';
        document.getElementById('adminTotalCount').textContent=rows.length;
        tbody.querySelectorAll('button').forEach(btn=>btn.addEventListener('click',async ()=>{
          if(confirm('确定取消该预约？')){await deleteSlot(btn.dataset.dk,btn.dataset.sid);renderScheduleTable();renderAdminTable();}
        }));
      }
      document.getElementById('btnExportCSV').addEventListener('click',()=>{
        const rows=[];getWeekDates(currentWeekMonday).forEach(d=>{const dk=formatDateKey(d);TIME_SLOTS.forEach(s=>{const data=getSlotData(dk,s.id);if(data)rows.push({日期:formatDateDisplay(d),星期:WEEKDAY_NAMES[d.getDay()],时段:s.label+' '+s.timeRange,微信号:data.wechat,出生:data.birthDate+' '+data.birthTime,地点:data.birthPlace,方式:data.online});})});
        if(rows.length===0) return showToast('无数据','info');
        const csv='\uFEFF'+Object.keys(rows[0]).join(',')+'\n'+rows.map(r=>Object.values(r).map(v=>`"${v}"`).join(',')).join('\n');
        const blob=new Blob([csv],{type:'text/csv'});
        const a=document.createElement('a');a.href=URL.createObjectURL(blob);a.download='预约导出.csv';a.click();
      });

      function showToast(msg,type){const t=document.createElement('div');t.className='toast '+type;t.textContent=msg;document.getElementById('toastContainer').appendChild(t);setTimeout(()=>t.remove(),2000);}

      // 初始化：拉取远程数据再渲染
      async function init(){
        try{
          await fetchRemoteData();
        }catch(e){
          showToast('加载数据失败，将使用空白表格','error');
          localCache={};
        }
        renderScheduleTable();
        document.getElementById('btnPrevWeek').addEventListener('click',()=>{currentWeekMonday.setDate(currentWeekMonday.getDate()-7);renderScheduleTable();});
        document.getElementById('btnNextWeek').addEventListener('click',()=>{currentWeekMonday.setDate(currentWeekMonday.getDate()+7);renderScheduleTable();});
        document.getElementById('btnToday').addEventListener('click',()=>{currentWeekMonday=getMondayOfWeek(new Date());renderScheduleTable();});
      }
      init();
    })();
  </script>
