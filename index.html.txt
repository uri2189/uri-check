    <script>
        (function() {
            // ========== 配置：替换成你自己的JSONBin信息 ==========
            const JSONBIN_BIN_ID = '你的BIN_ID';          // 例如 '64abc123...'
            const JSONBIN_API_KEY = '你的ACCESS_KEY';     // 在 jsonbin.io 的 API Keys 页面获取
            const JSONBIN_URL = `https://api.jsonbin.io/v3/b/${JSONBIN_BIN_ID}`;
            const ADMIN_PASSWORD = 'admin888';

            const TIME_SLOTS = [
                { id: 'morning', label: '上午', timeRange: '9:00-12:00', startHour: 9, icon: '☀️' },
                { id: 'afternoon', label: '下午', timeRange: '15:00-18:00', startHour: 15, icon: '🌤️' },
                { id: 'evening', label: '晚上', timeRange: '19:00-22:00', startHour: 19, icon: '🌙' },
            ];
            const WEEKDAY_NAMES = ['周日', '周一', '周二', '周三', '周四', '周五', '周六'];

            let currentWeekMonday = getMondayOfWeek(new Date());
            let selectedSlotKey = null;
            let isAdminLoggedIn = false;
            let localCache = {};

            function getMondayOfWeek(date) {
                const d = new Date(date);
                const day = d.getDay();
                const diff = day === 0 ? -6 : 1 - day;
                d.setDate(d.getDate() + diff);
                d.setHours(0, 0, 0, 0);
                return d;
            }
            function getWeekDates(monday) {
                const dates = [];
                for (let i = 0; i < 7; i++) {
                    const d = new Date(monday);
                    d.setDate(d.getDate() + i);
                    dates.push(d);
                }
                return dates;
            }
            function formatDateKey(date) {
                const y = date.getFullYear();
                const m = String(date.getMonth() + 1).padStart(2, '0');
                const d = String(date.getDate()).padStart(2, '0');
                return `${y}-${m}-${d}`;
            }
            function formatDateDisplay(date) {
                return `${date.getMonth() + 1}月${date.getDate()}日`;
            }
            function isSlotExpired(date, startHour) {
                const now = new Date();
                const slotDateTime = new Date(date);
                slotDateTime.setHours(startHour, 0, 0, 0);
                return now >= slotDateTime;
            }

            async function fetchRemoteData() {
                const resp = await fetch(JSONBIN_URL, {
                    headers: { 'X-Master-Key': JSONBIN_API_KEY }
                });
                if (!resp.ok) throw new Error('数据读取失败');
                const json = await resp.json();
                localCache = json.record;
                return localCache;
            }

            async function updateRemoteData(newData) {
                const resp = await fetch(JSONBIN_URL, {
                    method: 'PUT',
                    headers: {
                        'Content-Type': 'application/json',
                        'X-Master-Key': JSONBIN_API_KEY
                    },
                    body: JSON.stringify(newData)
                });
                if (!resp.ok) throw new Error('数据更新失败');
                localCache = newData;
            }

            function getSlotData(dateKey, slotId) {
                return localCache[dateKey]?.[slotId] || null;
            }

            async function bookSlot(dateKey, slotId, bookingInfo) {
                await fetchRemoteData(); // 拉取最新，防止覆盖
                if (getSlotData(dateKey, slotId)) throw new Error('时段已被占用');
                if (!localCache[dateKey]) localCache[dateKey] = {};
                localCache[dateKey][slotId] = bookingInfo;
                await updateRemoteData(localCache);
            }

            async function deleteSlot(dateKey, slotId) {
                await fetchRemoteData();
                if (localCache[dateKey] && localCache[dateKey][slotId]) {
                    delete localCache[dateKey][slotId];
                    if (Object.keys(localCache[dateKey]).length === 0) delete localCache[dateKey];
                    await updateRemoteData(localCache);
                }
            }

            function renderScheduleTable() {
                const weekDates = getWeekDates(currentWeekMonday);
                const startDisplay = formatDateDisplay(weekDates[0]);
                const endDisplay = formatDateDisplay(weekDates[6]);
                const year = weekDates[0].getFullYear();
                document.getElementById('weekLabel').textContent = `${year}年 · ${startDisplay} — ${endDisplay}`;

                let headerHTML = '<th style="width:70px;min-width:55px;">时段</th>';
                weekDates.forEach(d => {
                    const todayStr = formatDateKey(new Date());
                    const isTodayCol = formatDateKey(d) === todayStr;
                    headerHTML += `<th class="${isTodayCol ? 'today-col' : ''}">
                        <span class="date-num">${d.getDate()}</span>
                        <span class="date-month">${d.getMonth()+1}月 · ${WEEKDAY_NAMES[d.getDay()]}</span>
                    </th>`;
                });
                document.getElementById('tableHeaderRow').innerHTML = headerHTML;

                let bodyHTML = '';
                TIME_SLOTS.forEach(slot => {
                    bodyHTML += '<tr>';
                    bodyHTML += `<td class="time-label">${slot.icon} ${slot.label}<span class="time-range">${slot.timeRange}</span></td>`;
                    weekDates.forEach(d => {
                        const dateKey = formatDateKey(d);
                        const data = getSlotData(dateKey, slot.id);
                        const expired = isSlotExpired(d, slot.startHour);
                        let cls = '', icon = '', text = '';
                        if (expired) {
                            cls = 'expired'; icon = '⏰'; text = '已过期';
                        } else if (data) {
                            cls = 'booked'; icon = '🔒'; text = '已预约';
                        } else {
                            cls = 'free'; icon = '✨'; text = '点击预约';
                        }
                        bodyHTML += `<td class="slot-cell ${cls}" ${cls === 'free' ? `data-date-key="${dateKey}" data-slot-id="${slot.id}"` : ''}>
                            <span class="slot-icon">${icon}</span><span class="slot-status">${text}</span>
                        </td>`;
                    });
                    bodyHTML += '</tr>';
                });
                document.querySelector('#scheduleTable tbody').innerHTML = bodyHTML;

                document.querySelectorAll('.slot-cell.free').forEach(cell => {
                    cell.addEventListener('click', () => {
                        openBookingModal(cell.dataset.dateKey, cell.dataset.slotId);
                    });
                });
            }

            function openBookingModal(dateKey, slotId) {
                const existing = getSlotData(dateKey, slotId);
                if (existing) { showToast('该时段已被预约', 'error'); renderScheduleTable(); return; }
                const [y, m, d] = dateKey.split('-');
                const dateObj = new Date(y, m - 1, d);
                const slot = TIME_SLOTS.find(s => s.id === slotId);
                if (slot && isSlotExpired(dateObj, slot.startHour)) {
                    showToast('该时段已过期', 'error');
                    renderScheduleTable();
                    return;
                }
                selectedSlotKey = dateKey + '|' + slotId;
                document.getElementById('selectedSlotInfo').textContent = `📅 ${formatDateDisplay(dateObj)} ${WEEKDAY_NAMES[dateObj.getDay()]} · ${slot.icon} ${slot.label} ${slot.timeRange}`;
                document.getElementById('bookingForm').reset();
                document.getElementById('modalOverlay').style.display = 'flex';
            }

            document.getElementById('btnCancelBooking').addEventListener('click', () => {
                document.getElementById('modalOverlay').style.display = 'none';
                selectedSlotKey = null;
            });

            document.getElementById('bookingForm').addEventListener('submit', async function(e) {
                e.preventDefault();
                if (!selectedSlotKey) return;
                const [dateKey, slotId] = selectedSlotKey.split('|');
                const wechat = document.getElementById('wechat').value.trim();
                const birthDate = document.getElementById('birthDate').value;
                const birthTime = document.getElementById('birthTime').value;
                const birthPlace = document.getElementById('birthPlace').value.trim();
                const onlineRadio = document.querySelector('input[name="online"]:checked');
                const online = onlineRadio ? onlineRadio.value : null;

                if (!wechat || !birthDate || !birthTime || !birthPlace || !online) {
                    showToast('请完整填写所有信息', 'error');
                    return;
                }

                try {
                    await bookSlot(dateKey, slotId, {
                        wechat, birthDate, birthTime, birthPlace, online,
                        timestamp: new Date().toISOString()
                    });
                    document.getElementById('modalOverlay').style.display = 'none';
                    selectedSlotKey = null;
                    renderScheduleTable();
                    showToast('预约成功！', 'success');
                    if (isAdminLoggedIn && document.getElementById('adminContent').style.display !== 'none') {
                        renderAdminTable();
                    }
                } catch (err) {
                    showToast(err.message, 'error');
                }
            });

            document.getElementById('btnAdminTrigger').addEventListener('click', () => {
                document.getElementById('adminPanel').style.display = 'block';
                document.getElementById('adminContent').style.display = 'none';
                document.getElementById('adminPasswordInput').value = '';
                isAdminLoggedIn = false;
            });
            document.getElementById('btnAdminClose').addEventListener('click', () => {
                document.getElementById('adminPanel').style.display = 'none';
                document.getElementById('adminContent').style.display = 'none';
                isAdminLoggedIn = false;
            });
            document.getElementById('btnAdminLogin').addEventListener('click', () => {
                if (document.getElementById('adminPasswordInput').value === ADMIN_PASSWORD) {
                    isAdminLoggedIn = true;

                    renderAdminTable();
                    showToast('🔓 验证成功', 'success');
                } else {
                    showToast('密码错误', 'error');
                }
            });

            function renderAdminTable() {
                const weekDates = getWeekDates(currentWeekMonday);
                const rows = [];
                weekDates.forEach(d => {
                    const dateKey = formatDateKey(d);
                    const weekDay = WEEKDAY_NAMES[d.getDay()];
                    const dateDisplay = formatDateDisplay(d);
                    TIME_SLOTS.forEach(slot => {
                        const data = getSlotData(dateKey, slot.id);
                        if (data) {
                            rows.push({
                                dateDisplay,
                                weekDay,
                                slotLabel: slot.label,
                                slotTimeRange: slot.timeRange,
                                slotIcon: slot.icon,
                                dateKey,
                                slotId: slot.id,
                                ...data
                            });
                        }
                    });
                });
                const tbody = document.getElementById('adminTableBody');
                tbody.innerHTML = rows.length === 0 ?
                    '<tr><td colspan="9" style="padding:30px;color:#b0b0b0;">暂无预约记录</td></tr>' :
                    rows.map(r => `
                        <tr>
                            <td>${r.dateDisplay}</td>
                            <td>${r.weekDay}</td>
                            <td>${r.slotIcon} ${r.slotLabel}<br><small>${r.slotTimeRange}</small></td>
                            <td>${escapeHTML(r.wechat)}</td>
                            <td>${r.birthDate}</td>
                            <td>${r.birthTime}</td>
                            <td>${escapeHTML(r.birthPlace)}</td>
                            <td>${r.online === '线上' ? '🌐 线上' : '🏠 线下'}</td>
                            <td><button class="delete-btn" data-date-key="${r.dateKey}" data-slot-id="${r.slotId}">取消预约</button></td>
                        </tr>
                    `).join('');
                document.getElementById('adminTotalCount').textContent = rows.length;
                document.querySelectorAll('.delete-btn').forEach(btn => {
                    btn.addEventListener('click', async () => {
                        if (confirm('确定要取消该预约吗？')) {
                            await deleteSlot(btn.dataset.dateKey, btn.dataset.slotId);
                            renderScheduleTable();
                            renderAdminTable();
                            showToast('已取消预约', 'success');
                        }
                    });
                });
            }

            function escapeHTML(str) {
                const div = document.createElement('div');
                div.textContent = str;
                return div.innerHTML;
            }

            document.getElementById('btnExportCSV').addEventListener('click', () => {
                const weekDates = getWeekDates(currentWeekMonday);
                const rows = [];
                weekDates.forEach(d => {
                    const dateKey = formatDateKey(d);
                    TIME_SLOTS.forEach(slot => {
                        const data = getSlotData(dateKey, slot.id);
                        if (data) {
                            rows.push({
                                日期: formatDateDisplay(d),
                                星期: WEEKDAY_NAMES[d.getDay()],
                                时段: slot.label + ' ' + slot.timeRange,
                                微信号: data.wechat,
                                出生日期: data.birthDate,
                                出生时间: data.birthTime,
                                出生地点: data.birthPlace,
                                线上或线下: data.online
                            });
                        }
                    });
                });
                if (rows.length === 0) { showToast('暂无数据', 'info'); return; }
                const csv = '\uFEFF' + Object.keys(rows[0]).join(',') + '\n' +
                    rows.map(r => Object.values(r).map(v => `"${v}"`).join(',')).join('\n');
                const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
                const a = document.createElement('a');
                a.href = URL.createObjectURL(blob);
                a.download = `预约数据_${formatDateKey(currentWeekMonday)}.csv`;
                a.click();
                showToast('导出成功', 'success');
            });
